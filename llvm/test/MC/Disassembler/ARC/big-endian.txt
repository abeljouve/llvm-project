# RUN: FileCheck --check-prefix=ERR %s < %t.err

# On a big-endian target every 16-bit parcel is stored most significant byte
# first, so a 32-bit instruction or a long immediate is a plain big-endian
# word. The instruction length is taken from the first byte.

# 16-bit instructions.
# CHECK: nop_s
0x78 0xe0
# CHECK: mov_s %r1, 1
0xd9 0x01

# 32-bit instructions.
# CHECK: lsr %r15, %r1, 8
0x29 0x41 0x02 0x0f
# CHECK: mov.ne %r1, 10
0x21 0xca 0x02 0xa2
# CHECK: lr %r0, [33]
0x20 0xaa 0x08 0x40
# CHECK: lr %r0, [3]
0x20 0x6a 0x00 0xc0
# CHECK: ld %r0, [%r1,4]
0x11 0x04 0x00 0x00

# 16-bit instruction followed by a long immediate.
# CHECK: mov_s %r0, 305419896
0x40 0xc3 0x12 0x34 0x56 0x78

# 32-bit instructions followed by a long immediate.
# CHECK: mov %r2, 305419896
0x22 0x0a 0x0f 0x80 0x12 0x34 0x56 0x78
# CHECK: ld %r1, [305419896]
0x16 0x00 0x70 0x01 0x12 0x34 0x56 0x78
# CHECK: st %r1, [305419896]
0x1e 0x00 0x70 0x40 0x12 0x34 0x56 0x78
# CHECK: ld %r1, [%r0,305419896]
0x20 0x30 0x0f 0x81 0x12 0x34 0x56 0x78
